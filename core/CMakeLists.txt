# Copies known_dumps.json next to a shell executable (read by --identify and --dump).
function(mpt_copy_known_dumps target)
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MPT_KNOWN_DUMPS_JSON}"
            "$<TARGET_FILE_DIR:${target}>/data/known_dumps.json")
endfunction()

# Writes registered_titles.cpp for `target` from every mpt_register_title() call
# (P1-06 a) and links the registering libraries into it.
function(mpt_generate_title_registration target)
  get_property(_targets GLOBAL PROPERTY MPT_TITLE_TARGETS)
  get_property(_functions GLOBAL PROPERTY MPT_TITLE_FUNCTIONS)
  set(MPT_TITLE_DECLARATIONS "")
  set(MPT_TITLE_CALLS "")
  foreach(_fn IN LISTS _functions)
    string(APPEND MPT_TITLE_DECLARATIONS "void ${_fn}(mpt::shell::ShellTitles& titles);\n")
    string(APPEND MPT_TITLE_CALLS "  ::${_fn}(titles);\n")
  endforeach()
  set(_out "${CMAKE_CURRENT_BINARY_DIR}/generated/${target}_registered_titles.cpp")
  configure_file("${PROJECT_SOURCE_DIR}/core/shell/registered_titles.cpp.in" "${_out}" @ONLY)
  target_sources(${target} PRIVATE "${_out}")
  if(_targets)
    target_link_libraries(${target} PRIVATE ${_targets})
  endif()
  list(LENGTH _functions _count)
  message(STATUS "Shell title registrations for ${target}: ${_count}")
endfunction()
